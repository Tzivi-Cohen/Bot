Embedded Systems – יסודות Embedded C
=====================================

1. מה מייחד תכנות למערכות משובצות?
-----------------------------------
מערכת משובצת (Embedded System) היא מחשב שמובנה בתוך מוצר ומבצע משימה מוגדרת, למשל בקר של מכונת כביסה, חיישן תעשייתי או בקר מנוע. בהשוואה לתכנות למחשב רגיל:
- המשאבים מוגבלים. זיכרון ה־RAM הוא לעיתים כמה קילובייטים בלבד.
- לרוב אין מערכת הפעלה (Bare-Metal), או שיש מערכת הפעלה קטנה לזמן אמת (RTOS).
- הקוד ניגש ישירות לחומרה דרך רגיסטרים.
- יש דרישות תזמון, צריכת הספק ואמינות. הקושחה (Firmware) אמורה לרוץ שנים בלי אתחול.
- הקוד מקומפל במחשב הפיתוח עבור מעבד אחר (Cross-Compilation), למשל עם arm-none-eabi-gcc למעבדי ARM Cortex-M.

2. טיפוסים בגודל קבוע – stdint.h
--------------------------------
ב־C, הגודל של int תלוי בפלטפורמה: הוא 16 ביט ב־AVR של 8 ביט, ו־32 ביט ב־ARM Cortex-M. בקוד Embedded משתמשים בטיפוסים מ־<stdint.h>, שגודלם מובטח:
    uint8_t, int8_t      – 8 ביט
    uint16_t, int16_t    – 16 ביט
    uint32_t, int32_t    – 32 ביט
    uint64_t, int64_t    – 64 ביט
לרגיסטרים, לבתים ולמסכות ביטים משתמשים בטיפוסים בלי סימן (unsigned). הזזה של מספר שלילי ימינה תלויה במימוש, והזזה שלו שמאלה היא Undefined Behavior.

3. פעולות על ביטים (Bit Manipulation)
--------------------------------------
    #include <stdint.h>

    uint8_t reg = 0x00;

    reg |=  (1U << 3);     /* Set     – הדלקת ביט 3 */
    reg &= ~(1U << 3);     /* Clear   – כיבוי ביט 3 */
    reg ^=  (1U << 3);     /* Toggle  – היפוך ביט 3 */
    if (reg & (1U << 3)) { /* Test    – בדיקת ביט 3 */ }

מאקרואים נפוצים:
    #define BIT(n)            (1UL << (n))
    #define SET_BIT(r, n)     ((r) |=  BIT(n))
    #define CLEAR_BIT(r, n)   ((r) &= ~BIT(n))
    #define READ_BIT(r, n)    (((r) >> (n)) & 1UL)

עדכון שדה של כמה ביטים, למשל שדה של 2 ביטים בתחילת ביט 4:
    uint32_t value = 0xFFFFFFFFu;
    value = (value & ~(0x3UL << 4)) | ((2UL & 0x3UL) << 4);
    /* קודם מנקים את השדה, אחר כך כותבים את הערך החדש */

הסיומת U או UL חשובה. 1 << 31 על int בעל סימן של 32 ביט הוא Undefined Behavior, ואילו 1U << 31 מוגדר היטב.

4. volatile – למה ומתי
-----------------------
המילה volatile אומרת לקומפיילר שהערך יכול להשתנות "מבחוץ", בלי שהקוד הנוכחי שינה אותו. לכן הקומפיילר חייב לקרוא אותו מהזיכרון בכל גישה, ואסור לו לבצע עליו אופטימיזציות כמו שמירה ברגיסטר או מחיקה של קריאות "מיותרות".
משתמשים ב־volatile עבור:
1) רגיסטרים של חומרה (Memory-Mapped I/O).
2) משתנים שמשותפים ל־ISR ולקוד הראשי.
3) משתנים שמשותפים לכמה משימות (Threads/Tasks), אבל שם volatile לבדו אינו מספיק – צריך גם מנגנון סנכרון.

דוגמה לבאג שקורה בלי volatile:
    uint8_t data_ready = 0;            /* חסר volatile! */
    void UART_IRQHandler(void) { data_ready = 1; }
    int main(void) {
        while (!data_ready) { }        /* עם אופטימיזציה, הקומפיילר עלול להפוך זאת ללולאה אינסופית */
    }
התיקון: volatile uint8_t data_ready = 0;

volatile אינו הופך פעולה לאטומית. הפעולה count++ על משתנה volatile היא עדיין קריאה, שינוי וכתיבה – שלוש פעולות שפסיקה יכולה להיכנס ביניהן.

5. גישה לרגיסטרים לפי כתובת (Memory-Mapped I/O)
-----------------------------------------------
ציוד היקפי (Peripherals) של מיקרו־בקר נשלט דרך רגיסטרים שנמצאים בכתובות קבועות במרחב הזיכרון. הכתובות מופיעות במדריך העזר (Reference Manual) של המיקרו־בקר. ניגשים אליהן דרך מצביע volatile:

    #define REG32(addr)   (*(volatile uint32_t *)(addr))
    /* דוגמה עקרונית – הכתובת כאן היא דמה! יש לקחת כתובת אמיתית מה־Reference Manual */
    #define PERIPH_CTRL   REG32(0x40000000UL)

    PERIPH_CTRL |= BIT(0);    /* הפעלת ביט Enable */

בפועל, היצרן מספק קובצי Header (למשל לפי תקן CMSIS ב־ARM) שמגדירים מבנים (struct) לכל ציוד היקפי. כך כותבים GPIOA->ODR במקום לחשב כתובות.

6. const ושמירת נתונים בזיכרון Flash
------------------------------------
במיקרו־בקרים רבים, משתנה גלובלי שמוגדר const (למשל טבלת חיפוש) יכול להישמר בזיכרון Flash במקום ב־RAM, שהוא משאב יקר יותר. ב־ARM Cortex-M עם GCC זו ההתנהגות הרגילה. במיקרו־בקרי AVR (כמו ATmega328P) ה־Flash וה־RAM נמצאים במרחבי כתובות נפרדים, ולכן צריך מנגנון מיוחד: PROGMEM, ובסביבת Arduino גם המאקרו F() למחרוזות.

    static const uint16_t sine_table[8] = {
        2048, 3496, 4095, 3496, 2048, 600, 0, 600
    };

7. מכונת מצבים (State Machine)
------------------------------
קושחה רבה בנויה כמכונת מצבים: כל מצב מגדיר התנהגות, ואירועים גורמים למעבר בין מצבים. המבנה ברור, קל לבדיקה, ואינו חוסם את הלולאה הראשית.

    typedef enum { ST_IDLE, ST_HEATING, ST_DONE, ST_ERROR } State;

    State step(State s, int temp, int target, int sensor_ok) {
        if (!sensor_ok) return ST_ERROR;
        switch (s) {
            case ST_IDLE:    return ST_HEATING;
            case ST_HEATING: return (temp >= target) ? ST_DONE : ST_HEATING;
            case ST_DONE:    return ST_DONE;
            default:         return ST_ERROR;
        }
    }

8. מבנה טיפוסי של קושחת Bare-Metal – Super Loop
-----------------------------------------------
    int main(void) {
        system_init();          /* שעונים, GPIO, ציוד היקפי */
        while (1) {
            read_inputs();
            update_state_machine();
            write_outputs();
            /* אפשר להיכנס למצב חיסכון באנרגיה עד הפסיקה הבאה, למשל עם __WFI() ב־Cortex-M */
        }
    }
כאן main לעולם אינה מסתיימת, כי אין מערכת הפעלה לחזור אליה.

9. שאלות תרגול ופתרונות
------------------------
שאלה 1: כתבו ביטוי שבודק אם ביטים 0 ו־2 של reg דלוקים שניהם.
פתרון: (reg & 0x05) == 0x05. המסכה היא 0b101. צריך להשוות למסכה, כי בדיקה של (reg & 0x05) לבדה מתקיימת גם כשרק אחד מהביטים דלוק.

שאלה 2: למה משתמשים ב־uint32_t ולא ב־int לייצוג רגיסטר של 32 ביט?
פתרון: הגודל של int תלוי בפלטפורמה, ויש לו סימן. ‏uint32_t מבטיח 32 ביט בדיוק ובלי סימן, כך שהזזות ומסכות מתנהגות באופן מוגדר וצפוי.

שאלה 3: מה עלול לקרות אם משתנה שמשתנה בפסיקה אינו volatile?
פתרון: הקומפיילר עשוי לקרוא אותו פעם אחת ולשמור אותו ברגיסטר. אז הקוד הראשי לא יראה את העדכון מהפסיקה, ולולאת המתנה עלולה להפוך לאינסופית.

שאלה 4: מה הערך של x אחרי: uint8_t x = 0xF0; x ^= 0xFF; x &= ~(1U << 0);
פתרון: 0xF0 ^ 0xFF = 0x0F. אחר כך מכבים את ביט 0: ‏0x0F & 0xFE = 0x0E.
